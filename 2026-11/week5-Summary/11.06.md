# 内连接
# 隐式内连接
select e.name,d.name from emp e,dept d where e.dept_id=d.id;
# 显式内连接
select e.name,d.name from emp e join dept d on e.dept_id = d.id;

# 左连接
select e.*,d.name from emp e left join dept d on d.id = e.dept_id;
# 右连接，但其实右连接也能写成左连接，建议以后都用左连接吧.
select d.*,e.* from dept d right join emp e on d.id = e.dept_id;
select d.*,e.* from emp e left join dept d on d.id = e.dept_id;
# 自连接，我感觉其实就是给一个表创建了一个分身，然后SQL语句也是之前所写的左右连接，内连接。
