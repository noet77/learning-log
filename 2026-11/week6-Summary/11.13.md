## 这几天学了/做了什么

- JDBC其实就是用java操作数据库，首先就是加载驱动驱动类，然后就是获取数据库连接，最后就是执行sql语句，输出结果，关闭连接。导了一下Druid连接池，尚硅谷的课后面给我看的有点懵懵的，dao还没学，后面去做了个黑马的练习，大致逻辑已经明白了。

今天在学习 JDBC 批量操作时，原本的常规做法是‘单兵作战’，一条一条地发送 SQL 请求，结果耗时将近 1910ms，后来用了 Batch 批量处理，逻辑就像把所有货物打包成一个大集装箱，一次性发往数据库。看着控制台打印出的 138ms，通过这次对比我才明白，减少网络 I/O 的次数有多重要。

-package com.itheima.example;

import com.alibaba.druid.pool.DruidDataSourceFactory;
import com.itheima.pojo.Brand;
import org.junit.Test;

import javax.sql.DataSource;
import java.io.FileInputStream;
import java.sql.Connection;
import java.sql.PreparedStatement;
import java.sql.ResultSet;
import java.util.ArrayList;
import java.util.List;
import java.util.Properties;

public class BrandTest {
    /*
     * 查询所有
     * 1.SQL：select * from t_brand
     * 2.参数：无
     * 3.结果：List<Brand>
     */
    @Test
    public void testSelectAll()throws Exception{
        Properties prop = new Properties();
        prop.load(new FileInputStream("/Users/noetkk/Downloads/java/JDBC/jdbc-demo/src/db.properties"));
        //4.获取连接池
        DataSource dataSource = DruidDataSourceFactory.createDataSource(prop);
        //5.获取连接
        Connection conn = dataSource.getConnection();

        //定义sql
        String sql = "SELECT * FROM tb_brand;";
        PreparedStatement pstmt = conn.prepareStatement(sql);
        ResultSet rs = pstmt.executeQuery();
        List<Brand> brands = new ArrayList<>();
        while(rs.next()){
            int brandId = rs.getInt("id");
            String brandName = rs.getString("brand_name");
            String brandDesc = rs.getString("company_name");
            int ordered = rs.getInt("ordered");
            String description = rs.getString("description");
            int status = rs.getInt("status");
            //封装对象
            Brand brand = new Brand(brandId, brandName, brandDesc, ordered, description, status);
            brands.add(brand);
        }
        System.out.println(brands);
        rs.close();
        pstmt.close();
        conn.close();
   }


    /*
     * 添加
     * 1.SQL：insert into tb_brand (brand_name, company_name, ordered, description, status)values (?,?,?,?);
     * 2.参数：需要，除了id以外的所有参数
     * 3.结果：boolean
     */
    @Test
    public void testInsert()throws Exception{
        //接收前端页面传回来的信息
        String brandName = "星巴克";
        String brandDesc = "星巴克公司";
        int ordered = 10;
        String description = "一家专注高端服务的咖啡店";
        int status = 1;



        Properties prop = new Properties();
        prop.load(new FileInputStream("/Users/noetkk/Downloads/java/JDBC/jdbc-demo/src/db.properties"));
        //4.获取连接池
        DataSource dataSource = DruidDataSourceFactory.createDataSource(prop);
        //5.获取连接
        Connection conn = dataSource.getConnection();

        //定义sql
        String sql = "insert into tb_brand (brand_name, company_name, ordered, description, status)values (?,?,?,?,?);";
        //获取pstmt对象
        PreparedStatement pstmt = conn.prepareStatement(sql);
        //设置参数
        pstmt.setString(1, brandName);
        pstmt.setString(2, brandDesc);
        pstmt.setInt(3, ordered);
        pstmt.setString(4, description);
        pstmt.setInt(5, status);
        //执行sql
        int count=pstmt.executeUpdate();//影响行数
        //处理结果
        System.out.println(count>0);//输出布尔值
        pstmt.close();
        conn.close();
    }

    /*
     * 修改
     * 1.SQL：
     * update tb_brand set
           brand_name=?,
           company_name=?,
           ordered=?,
           description=?,
           status=?
       where id=?
     * 2.参数：需要所有参数
     * 3.结果：boolean
     */

    public void testUpdate()throws Exception{
        //接收前端页面传回来的信息
        String brandName = "星巴克";
        String brandDesc = "星巴克公司";
        int ordered = 100;
        String description = "一家专注高端服务的咖啡店,很贵";
        int status = 1;

        Properties prop = new Properties();

        DataSource dataSource = DruidDataSourceFactory.createDataSource(prop);
        //5.获取连接
        Connection conn = dataSource.getConnection();

        //定义sql

        PreparedStatement pstmt = conn.prepareStatement(sql);
